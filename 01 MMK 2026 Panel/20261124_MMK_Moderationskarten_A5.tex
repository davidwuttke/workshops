% Header band: #1 kicker (left), #2 time (right), #3 name or title, #4 role or subtitle
\newcommand{\cardhead}[4]{%
\noindent
\begin{tikzpicture}
\node[fill=primaryBlue, text=white, inner xsep=4mm, inner ysep=3.2mm, outer sep=0pt,
      text width=\dimexpr\linewidth-8mm\relax, align=left] {%
  {\fontsize{9}{11}\selectfont\bfseries #1\hfill #2\par}\vspace{1.6mm}
  {\fontsize{21}{24}\selectfont\bfseries #3\par}\vspace{0.8mm}
  {\fontsize{12}{15}\selectfont #4\par}};
\end{tikzpicture}\par}

% Small blue label above each block
\newcommand{\lbl}[1]{\par\vspace{4.2mm}%
  {\fontsize{8.5}{10}\selectfont\bfseries\color{primaryBlue}\MakeUppercase{#1}\par}\vspace{1.4mm}}

% Two or three facts about the speaker
\newenvironment{facts}{%
  \begin{itemize}[leftmargin=4.5mm, labelsep=2mm, itemsep=1.6mm, topsep=0mm, parsep=0mm,
                  partopsep=0mm, label={\color{primaryBlue}\textbullet}]%
  \fontsize{13}{16.5}\selectfont}{\end{itemize}}

% Question to the room: #1 question, #2 what to ask hands for
\newcommand{\ask}[2]{%
\par\noindent
\begin{tikzpicture}
\node[fill=primaryBlue!8, inner xsep=4mm, inner ysep=3.6mm, outer sep=0pt,
      text width=\dimexpr\linewidth-8mm\relax, align=left] (q) {%
  {\fontsize{15.5}{20}\selectfont #1\par}\vspace{2.6mm}
  {\fontsize{11.5}{14}\selectfont\bfseries\color{primaryBlue}#2\par}};
\draw[primaryBlue, line width=1.4mm] ([xshift=0.7mm]q.north west) -- ([xshift=0.7mm]q.south west);
\end{tikzpicture}\par}

% The line that hands over to the speaker
\newcommand{\handover}[1]{\par{\fontsize{17}{21.5}\selectfont\bfseries #1\par}}

% Plain note, e.g. what to listen for
\newcommand{\note}[1]{\par{\fontsize{12}{15.5}\selectfont #1\par}}

% Footer and page break
\newcommand{\cardfoot}[1]{\par\vfill
  {\color{secondaryGray}\rule{\linewidth}{0.4pt}\par\vspace{1.2mm}
   \fontsize{9}{11}\selectfont #1\par}\newpage}

% Thesis as on the slide: #1 colour, #2 letter, #3 text
\newcommand{\thesis}[3]{%
\par\noindent
\begin{tikzpicture}
\node[fill=#1!10, inner xsep=3.2mm, inner ysep=2.8mm, outer sep=0pt, anchor=north west,
      text width=\dimexpr\linewidth-9mm-6.4mm\relax, align=left,
      font=\fontsize{12}{15.5}\selectfont] (t) at (9mm,0) {#3};
\fill[#1] (0,0) rectangle (t.south west);
\node[text=white, inner sep=0pt, font=\bfseries\fontsize{17}{17}\selectfont]
      at ($(0,0)!0.5!(t.south west)$) {#2};
\end{tikzpicture}\par\vspace{1.6mm}}

% Question to one panelist on a pair card: #1 name, #2 question
\newcommand{\q}[2]{\par\vspace{2.6mm}%
  {\fontsize{12}{15.5}\selectfont{\bfseries\color{primaryBlue}#1}\enspace #2\par}}

\cardhead{CARD 4 OF 5 \textbar{} PAIR 1 \textbar{} SLIDE 1/2}{14:32 \textbar{} 6 MIN}{The boomers}{What is industrial AI for?}

\vspace{4mm}
\thesis{primaryBlue}{A}{If industrial AI helps, it is because we can finally afford \textbf{fewer expensive skilled workers}.}
\thesis{tertiaryOrange}{B}{If industrial AI helps, it is because it catches the \textbf{boomers' knowledge} before it retires.}

\lbl{How}
\note{Hands in the room: A or B? Then \textbf{Ahr\then Engel\then Grunwald}.\\
A or B, one reason, one minute. No ``both are right''.}

\lbl{One question each}
\q{Ahr}{One humanoid for 2.5 workers. But your robots learn from today's skilled workers. So: replace them, or learn from them before they retire?}
\q{Engel}{The Cutting Assistant needs no cutting experience. Is that A or B: fewer experienced operators, or keeping what they know?}
\q{Grunwald}{30 percent leaner, 1,000 AI experts in Berlin. Fewer expensive engineers, or knowledge brought back in-house?}

\lbl{If time}
\note{To Ahr: the boomers are gone by 2037, your robots reach 50 percent after 2040. Who runs the lines in between?}

\cardhead{CARD 5 OF 5 \textbar{} PAIR 2 \textbar{} SLIDE 2/2}{14:38 \textbar{} 6 MIN}{Gen Z}{What does industrial AI do to them?}

\vspace{4mm}
\thesis{primaryBlue}{A}{For Gen Z, industrial AI is a \textbf{shortcut} to expertise. They start where the boomers ended.}
\thesis{tertiaryOrange}{B}{For Gen Z, industrial AI is a \textbf{substitute} for expertise. They never have to learn what the machine already knows and will never become as much experts as boomers are.}

\lbl{How}
\note{Hands in the room: A or B? Then \textbf{Engel\then Grunwald\then Ahr}.}

\q{Grunwald}{If AI writes the simple code, who becomes your next architect?}
\q{Ahr}{Your roadmap reads like an apprenticeship: the robot takes the simple tasks first. Where does someone who starts in 2030 learn to handle a disruption?}
