% Title-page metadata helpers ---------------------------------------------------
% Beamer already has the right machinery for multi-author title data:
%   \author[Short]{Alice \inst{1} \and Bob \inst{2}}

% The custom TUM/MGT title pages below now use the same Beamer insertions, rather

\makeatletter

% title pages and in short author/footer strings.

\makeatother

\newcommand{\TumTitleInstituteNormalized}{}
\ExplSyntaxOn
\tl_new:N \g_tum_title_institute_tl
\cs_new_protected:Npn \tum_store_title_institute:n #1
  {
    \tl_gset:Nn \g_tum_title_institute_tl {#1}
    % expl3 treats ~ as a normal space token here.  This replaces both
    % " | " and bare "|" with Beamer's standard institute separator.
    \tl_greplace_all:Nnn \g_tum_title_institute_tl {~|~} { \and }
    \tl_greplace_all:Nnn \g_tum_title_institute_tl {|} { \and }
    \cs_gset_protected:Npx \TumTitleInstituteNormalized
      { \exp_not:V \g_tum_title_institute_tl }
  }
\NewDocumentCommand{\TumStoreTitleInstitute}{+m}
  { \tum_store_title_institute:n {#1} }
\ExplSyntaxOff

\RenewDocumentCommand{\institute}{o +m}{%
  \TumStoreTitleInstitute{#2}%
  \IfNoValueTF{#1}{%
    \TumOriginalInstitute{\TumTitleInstituteNormalized}%
  }{%
    \TumOriginalInstitute[#1]{\TumTitleInstituteNormalized}%
  }%
}
